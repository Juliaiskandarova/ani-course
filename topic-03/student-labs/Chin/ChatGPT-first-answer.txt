// Предмет может входить в несколько специальностей. Тема принадлежит одному
// предмету. Вопрос может входить в несколько тем, в том числе разных предметов.
// При генерации темы выбираются из предметов специальности группы, а вопросы —
// из объединения выбранных тем. Один вопрос включается в вариант максимум один раз.

Enum test_variant_mode {
  common
  individual
}

Enum answer_outcome {
  correct
  partial
  incorrect
}

Table specialties {
  id bigint [pk, increment]
  code varchar(32) [not null, unique]
  name varchar(255) [not null]
}

Table subjects {
  id bigint [pk, increment]
  name varchar(255) [not null]
  deleted_at timestamptz
}

Table specialty_subjects {
  specialty_id bigint [not null]
  subject_id bigint [not null]

  indexes {
    (specialty_id, subject_id) [pk]
  }
}

Table topics {
  id bigint [pk, increment]
  subject_id bigint [not null]
  name varchar(255) [not null]
  deleted_at timestamptz
}

Table questions {
  id bigint [pk, increment]
  deleted_at timestamptz
}

Table topic_questions {
  topic_id bigint [not null]
  question_id bigint [not null]

  indexes {
    (topic_id, question_id) [pk]
  }
}

// Допущение: вопрос допускает выбор одного или нескольких вариантов ответа.
// У каждой редакции должен быть хотя бы один правильный вариант.
// Текст, варианты и ключ ответа выданной редакции сохраняются.
// При изменении создаётся новая редакция. Удаление из справочника логическое.

Table question_revisions {
  id bigint [pk, increment]
  question_id bigint [not null]
  revision_number int [not null]
  question_text text [not null]

  indexes {
    (question_id, revision_number) [unique]
  }
}

Table question_options {
  question_revision_id bigint [not null]
  option_number int [not null]
  option_text text [not null]
  is_correct boolean [not null]

  indexes {
    (question_revision_id, option_number) [pk]
  }
}

Table student_groups {
  id bigint [pk, increment]
  specialty_id bigint [not null]
  code varchar(32) [not null, unique]
}

Table students {
  id bigint [pk, increment]
  group_id bigint [not null]
  student_number varchar(32) [not null, unique]
  full_name varchar(255) [not null]
}

// Один тест — одна выдача для одной группы с общим сроком сдачи.
// При выдаче состав группы фиксируется строками test_attempts.
// При переводе студента его прошлые результаты остаются за группой из tests.
// После выдачи сохраняются параметры оценивания, темы и состав вариантов.
// archived_at отмечает перенос теста в архив. Результаты хранятся в тех же таблицах.

Table tests {
  id bigint [pk, increment]
  group_id bigint [not null]
  title varchar(255) [not null]
  variant_mode test_variant_mode [not null]
  issued_at timestamptz
  due_at timestamptz [not null]
  initial_points decimal(8,1) [not null]
  archived_at timestamptz

  indexes {
    (group_id, issued_at)
  }
}

Table test_topics {
  test_id bigint [not null]
  topic_id bigint [not null]

  indexes {
    (test_id, topic_id) [pk]
  }
}

// В режиме common всем студентам назначается один вариант.
// В режиме individual каждому назначается свой вариант. Наборы исходных
// вопросов должны различаться независимо от порядка и номера редакции.

Table test_variants {
  test_id bigint [not null]
  variant_number int [not null]

  indexes {
    (test_id, variant_number) [pk]
  }
}

Table test_variant_questions {
  test_id bigint [not null]
  variant_number int [not null]
  question_revision_id bigint [not null]
  position int [not null]

  indexes {
    (test_id, variant_number, question_revision_id) [pk]
    (test_id, variant_number, position) [unique]
  }
}

// Допущение: для каждого теста задаются пороги оценок 2, 3, 4 и 5.
// Порог для 2 равен нулю. Остальные пороги возрастают с оценкой.

Table test_grade_thresholds {
  test_id bigint [not null]
  grade smallint [not null]
  minimum_points decimal(8,1) [not null]

  indexes {
    (test_id, grade) [pk]
    (test_id, minimum_points) [unique]
  }
}

// Допущение: у студента одна попытка на выданный тест.
// Точное совпадение с ключом даёт correct. Непустое подмножество правильных
// вариантов даёт partial. Пустой ответ или выбор ошибочного варианта даёт incorrect.
// Из initial_points вычитается 1 за incorrect, 0.5 за partial и однократно
// 0.5 за сдачу после due_at. Итог ограничивается снизу нулём.
// Оценка определяется по наибольшему достигнутому порогу.
// При сдаче сохраняются исход каждого вопроса, итоговый балл и оценка.
// Для отчёта за период используется submitted_at. Двоечники — студенты
// с оценкой 2 за сданный тест в периоде. У несданных тестов оценка отсутствует.

Table test_attempts {
  test_id bigint [not null]
  student_id bigint [not null]
  variant_number int [not null]
  submitted_at timestamptz
  final_points decimal(8,1)
  grade smallint

  indexes {
    (test_id, student_id) [pk]
    (test_id, student_id, variant_number) [unique]
    (submitted_at)
  }
}

// Сложность вопроса можно оценить по доле неполных и неправильных ответов
// среди всех предъявлений в сданных тестах. Редакции объединяются по question_id.

Table test_answers {
  test_id bigint [not null]
  student_id bigint [not null]
  variant_number int [not null]
  question_revision_id bigint [not null]
  outcome answer_outcome

  indexes {
    (test_id, student_id, question_revision_id) [pk]
  }
}

Table test_answer_options {
  test_id bigint [not null]
  student_id bigint [not null]
  question_revision_id bigint [not null]
  option_number int [not null]

  indexes {
    (test_id, student_id, question_revision_id, option_number) [pk]
  }
}

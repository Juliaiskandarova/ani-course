В исправленной версии устранены избыточные поля состояния и добавлены механизмы сохранения истории. В частности, штраф и возмещение объединены в одну таблицу `loss_resolution`, а ограничение трёх одновременно выданных книг обеспечивается отдельной таблицей активных выдач с тремя слотами.

Межтабличные проверки, агрегатные ограничения и защита от конкурентных операций требуют триггеров и транзакционных проверок PostgreSQL. Они отмечены комментариями непосредственно в DBML. Локальные ограничения заданы через `CHECK`.

```
// =====================================================
// ACQUISITIONS
// =====================================================

// Документы поступления книг в библиотечный фонд.
//
// Проведенные документы неизменяемы.
// Для исправления используются отдельные корректировки.
//
// Для первоначального фонда используется opening_balance.
// Для бесплатных поступлений unit_price = 0,
// если учетная стоимость не определяется.
//
// Для оцененных пожертвований и первоначального фонда
// можно указать учетную стоимость экземпляра.

Table acquisition {
  acquisition_id bigint [pk, increment]

  document_number varchar(100) [not null, unique]

  acquired_at timestamptz [not null]
  acquisition_type acquisition_type [not null]

  status acquisition_status [not null, default: 'draft']

  notes text

  indexes {
    (acquired_at)
    (status, acquired_at)
  }

    `((status = 'posted' AND posted_at IS NOT NULL) OR (status IN ('draft', 'cancelled') AND posted_at IS NULL))` [name: 'chk_acquisition_posting']
  }
}

// Позиции документов приобретения.

// quantity и unit_price сохраняются как первоначальные

Table acquisition_item {
  acquisition_item_id bigint [pk, increment]

  acquisition_id bigint [not null]
  edition_id bigint [not null]

  quantity integer [not null]
  unit_price numeric(14,2) [not null]

  notes text

  indexes {
    (acquisition_id)
    (edition_id)
    (acquisition_item_id, edition_id) [unique]
  }

  checks {
    `quantity > 0` [name: 'chk_acquisition_quantity']
    `unit_price >= 0` [name: 'chk_acquisition_unit_price']
  }
}

// =====================================================
// ACQUISITION ADJUSTMENTS
// =====================================================

// Корректировки проведенных документов приобретения.
//
// quantity_delta и unit_price_delta являются знаковыми
// изменениями относительно предыдущих учетных значений.
//
// amount_delta =
//   new_quantity * new_unit_price
//   - old_quantity * old_unit_price.
//
// Отрицательные значения допустимы именно для корректировок.
// Итоговое количество, цена и стоимость не могут быть отрицательными.
//
// Корректировки неизменяемы после регистрации.
// Для отмены корректировки создается обратная корректировка.

Table acquisition_adjustment {
  adjustment_id bigint [pk, increment]

  acquisition_item_id bigint [not null]

  document_number varchar(100) [not null]
  line_number smallint [not null]

  posted_at timestamptz [not null]

  quantity_delta integer [not null, default: 0]
  unit_price_delta numeric(14,2) [not null, default: 0]
  amount_delta numeric(14,2) [not null]

  reason text [not null]

  indexes {
    (document_number, line_number) [unique]
    (acquisition_item_id, posted_at)
  }

  checks {
    `line_number > 0` [name: 'chk_adjustment_line_number']
    `quantity_delta <> 0 OR unit_price_delta <> 0` [name: 'chk_adjustment_has_change']
  }
}

    (copy_id, edition_id) [unique]

  checks {

  }

// =====================================================
// LOANS
// =====================================================

// История выдач.
//
// loan не содержит отдельного status:
// - returned_at IS NOT NULL — книга возвращена;
// - lost_at IS NOT NULL — книга утеряна;
// - оба NULL — выдача остается открытой.
//
// Активные выдачи дополнительно представлены в active_loan.
//
// book_id_at_issue, book_title_at_issue,
// genre_id_at_issue и genre_name_at_issue
// сохраняют характеристики книги на момент выдачи.

Table loan {
  loan_id bigint [pk, increment]

  member_id bigint [not null]

  copy_id bigint [not null]
  edition_id bigint [not null]

  book_id_at_issue bigint [not null]
  book_title_at_issue varchar(500) [not null]

  genre_id_at_issue bigint [not null]
  genre_name_at_issue varchar(120) [not null]

  issued_at timestamptz [not null]
  due_at timestamptz [not null]

  returned_at timestamptz
  lost_at timestamptz

  notes text

  indexes {
    (member_id, due_at)
    (copy_id, issued_at)
    (issued_at)
    (returned_at)
    (book_id_at_issue, issued_at)
    (genre_id_at_issue, issued_at)

    (loan_id, copy_id) [unique]
    (loan_id, copy_id, edition_id) [unique]
    (loan_id, copy_id, member_id) [unique]
  }

  checks {
    `due_at > issued_at` [name: 'chk_loan_due_date']
    `returned_at IS NULL OR returned_at >= issued_at` [name: 'chk_loan_return_date']
    `lost_at IS NULL OR lost_at >= issued_at` [name: 'chk_loan_lost_date']
    `returned_at IS NULL OR lost_at IS NULL` [name: 'chk_loan_return_or_loss']
  }
}

// ACTIVE LOANS

// Таблица содержит только текущие активные выдачи.

// =====================================================

// История продлений.

// renewal_number ограничен значениями 1-3.

  }

  checks {

// =====================================================
// OVERDUE NOTICES
// =====================================================

// История сформированных уведомлений.
//
// Один займ может иметь несколько уведомлений.
// Уникальность loan_id отсутствует намеренно.

Table overdue_notice {

  recipient_address_snapshot text
  recipient_email_snapshot varchar(254)

  body_text text [not null]

  sent_at timestamptz

  indexes {

  }

  checks {
    `sent_at IS NULL OR sent_at >= generated_at` [name: 'chk_notice_sent_date']
  }
}

// =====================================================
// LOST BOOK RESOLUTION
// =====================================================

// Общая таблица для двух взаимоисключающих операций:
//
// 1. fine — формирование штрафной квитанции.
//    Штраф автоматически считается оплаченным
//    в момент fine_assessed_at.
//
// 2. replacement — заявление о возмещении стоимости
//    утерянной книги другим экземпляром того же издания.
//
// Одна запись на один потерянный экземпляр/займ.
//
// Наличие resolution_type исключает одновременное
// оформление штрафа и возмещения.
//
// Для replacement:
// - отсутствие replacement_copy_id означает ожидание;
// - наличие replacement_copy_id и replacement_received_at
//   означает завершенное возмещение.

Table loss_resolution {
  resolution_id bigint [pk, increment]

  loan_id bigint [not null, unique]

  original_copy_id bigint [not null]
  edition_id bigint [not null]

  resolution_type loss_resolution_type [not null]

  // Штрафная квитанция

  fine_receipt_number varchar(50) [unique]

  edition_price_at_assessment numeric(14,2)
  fine_amount numeric(14,2)

  fine_assessed_at timestamptz
  fine_receipt_text text

  // Заявление о возмещении

  replacement_statement_number varchar(100) [unique]

  application_submitted_at timestamptz
  application_text text

  replacement_copy_id bigint [unique]
  replacement_received_at timestamptz

  notes text

  indexes {
    (resolution_type, application_submitted_at)
    (fine_assessed_at)
  }

  checks {
    `((resolution_type = 'fine' AND fine_receipt_number IS NOT NULL AND edition_price_at_assessment IS NOT NULL AND fine_amount IS NOT NULL AND fine_assessed_at IS NOT NULL AND fine_receipt_text IS NOT NULL AND replacement_statement_number IS NULL AND application_submitted_at IS NULL AND application_text IS NULL AND replacement_copy_id IS NULL AND replacement_received_at IS NULL) OR (resolution_type = 'replacement' AND fine_receipt_number IS NULL AND edition_price_at_assessment IS NULL AND fine_amount IS NULL AND fine_assessed_at IS NULL AND fine_receipt_text IS NULL AND replacement_statement_number IS NOT NULL AND application_submitted_at IS NOT NULL AND application_text IS NOT NULL AND ((replacement_copy_id IS NULL AND replacement_received_at IS NULL) OR (replacement_copy_id IS NOT NULL AND replacement_received_at IS NOT NULL))))` [name: 'chk_loss_resolution_type']

    `fine_amount IS NULL OR fine_amount = 3 * edition_price_at_assessment` [name: 'chk_fine_amount']

    `edition_price_at_assessment IS NULL OR edition_price_at_assessment >= 0` [name: 'chk_fine_edition_price']

    `fine_amount IS NULL OR fine_amount >= 0` [name: 'chk_fine_nonnegative']

    `replacement_copy_id IS NULL OR replacement_copy_id <> original_copy_id` [name: 'chk_replacement_not_original']

    `replacement_received_at IS NULL OR replacement_received_at >= application_submitted_at` [name: 'chk_replacement_received_date']
  }
}

// =====================================================
// INVENTORY WRITEOFFS
// =====================================================

// История списания физических экземпляров.
//
// copy_id уникален: один экземпляр списывается один раз.
//
// loan_id необязателен для книг, утерянных непосредственно
// в библиотечном фонде.
//
// Если списывается книга, выданная читателю,
// составной FK гарантирует соответствие займа экземпляру.
//
// Списание не закрывает задолженность читателя:
// обязательство определяется по loan и loss_resolution.

Table inventory_writeoff {
  writeoff_id bigint [pk, increment]

  copy_id bigint [not null, unique]
  loan_id bigint [unique]

  written_off_at timestamptz [not null]

  reason writeoff_reason [not null]

  act_number varchar(100) [not null, unique]

  notes text

  indexes {
    (written_off_at)
    (reason, written_off_at)
  }

  checks {
    `reason <> 'not_returned' OR loan_id IS NOT NULL` [name: 'chk_writeoff_unreturned_loan']
  }
}

// =====================================================
// BUSINESS RULES AND REQUIRED DATABASE TRIGGERS
// =====================================================

// -----------------------------------------------------
// 1. ACQUISITION DOCUMENTS AND COPY COUNTS
// -----------------------------------------------------

// При проведении acquisition:
//
// Для каждой acquisition_item проверяется:
//
// COUNT(book_copy WHERE registration_cancelled_at IS NULL)
//     = acquisition_item.quantity
//
// Все физические экземпляры должны быть зарегистрированы
// до проведения документа.
//
// После проведения acquisition запрещено:
// - изменять acquisition_item.quantity;
// - изменять acquisition_item.unit_price;
// - удалять позиции;
// - удалять зарегистрированные экземпляры.
//
// Количество экземпляров и стоимость могут исправляться
// только через acquisition_adjustment.
//
// Эффективное количество:
//
// acquisition_item.quantity
//     + SUM(acquisition_adjustment.quantity_delta)
//
// Эффективная цена экземпляра:
//
// acquisition_item.unit_price
//     + SUM(acquisition_adjustment.unit_price_delta)
//
// При проведении корректировки проверяется:
//
// 1. Эффективное количество >= 0.
// 2. Эффективная цена >= 0.
// 3. Эффективная стоимость строки >= 0.
// 4. amount_delta равен разнице новой и предыдущей
//    полной стоимости строки.
// 5. Количество незакрытых регистраций book_copy
//    соответствует эффективному количеству.
//
// При quantity_delta > 0 новые записи book_copy
// создаются в той же транзакции.
//
// При quantity_delta < 0 соответствующие ошибочные
// регистрации помечаются registration_cancelled_at.
//
// Отменять регистрацию экземпляров, по которым уже
// были выдачи, списания или возмещения, запрещено.
//
// Потерянные и списанные экземпляры не исключаются
// из исторического количества приобретенных книг.
//
// Все корректировки являются неизменяемыми.
// Для отмены корректировки создается обратная запись.
//
// -----------------------------------------------------
// 2. СОСТОЯНИЕ ФИЗИЧЕСКИХ ЭКЗЕМПЛЯРОВ
// -----------------------------------------------------

// Отдельное поле book_copy.status отсутствует.
//
// Состояние экземпляра определяется следующим образом:
//
// 1. registration_cancelled_at IS NOT NULL —
//    ошибочная регистрация отменена.
//
// 2. Есть inventory_writeoff — экземпляр списан.
//
// 3. Есть loan.lost_at — экземпляр утерян.
//
// 4. Есть active_loan — экземпляр выдан.
//
// 5. В остальных случаях — экземпляр доступен.
//
// При выдаче запрещено использовать экземпляр:
// - с отмененной регистрацией;
// - с историей потери;
// - со списанием;
// - с активной выдачей.
//
// Эти проверки выполняются триггером.

// -----------------------------------------------------
// 3. СОГЛАСОВАННОСТЬ LOAN И ACTIVE_LOAN
// -----------------------------------------------------

// active_loan содержит только открытые выдачи.
//
// Для каждого займа, у которого:
// returned_at IS NULL AND lost_at IS NULL,
// должна существовать ровно одна запись active_loan.
//
// Для каждого займа с заполненным returned_at или lost_at
// запись active_loan должна отсутствовать.
//
// Создание выдачи:
// - создание loan;
// - создание active_loan;
// - фиксация исторических авторов;
// - фиксация жанра и названия;
// выполняются одной транзакцией.
//
// Возврат:
// - заполнение loan.returned_at;
// - удаление active_loan.
//
// Потеря:
// - заполнение loan.lost_at;
// - удаление active_loan.
//
// Согласованность проверяется отложенными триггерами
// перед завершением транзакции.
//
// -----------------------------------------------------
// 4. ОГРАНИЧЕНИЯ ПРИ ВЫДАЧЕ
// -----------------------------------------------------

// Таблица active_loan обеспечивает:
//
// - максимум три активные выдачи на одного читателя;
// - максимум одну активную выдачу на экземпляр.
//
// member_slot принимает только значения 1, 2, 3.
//
// Уникальность (member_id, member_slot) не позволяет
// занять один слот двумя выдачами.
//
// Уникальность copy_id не позволяет выдать один экземпляр
// нескольким читателям одновременно.
//
// Перед созданием active_loan триггер должен:
//
// 1. Заблокировать строку library_member
//    через SELECT FOR UPDATE.
//
// 2. Проверить, что последний статус читателя — active.
//
// 3. Проверить отсутствие просроченных активных выдач:
//
//    active_loan JOIN loan
//    WHERE loan.due_at < CURRENT_TIMESTAMP.
//
// 4. Проверить отсутствие неурегулированных потерь:
//
//    - потерянных книг без loss_resolution;
//    - заявлений replacement без replacement_received_at.
//
// 5. Проверить доступность экземпляра.
//
// 6. Проверить возможность занятия выбранного member_slot.
//
// Блокировка строки читателя сериализует операции выдачи
// одного пользователя, выполняемые разными сотрудниками.
//
// UNIQUE-ограничения остаются дополнительной защитой
// от конкурентных операций над одним экземпляром
// или одним слотом.
//
// -----------------------------------------------------
// 5. ПРОДЛЕНИЕ ВЫДАЧ
// -----------------------------------------------------

// При добавлении loan_renewal триггер проверяет:
//
// - наличие активной выдачи;
// - renewal_number = предыдущий номер + 1;
// - renewal_number не больше 3;
// - previous_due_at = текущему loan.due_at;
// - new_due_at > previous_due_at;
// - renewed_at не раньше issued_at.
//
// После успешного добавления записи:
//
// loan.due_at = loan_renewal.new_due_at.
//
// Таким образом, loan.due_at всегда содержит
// текущий срок возврата.
//
// История продлений неизменяема.
//
// -----------------------------------------------------
// 6. ШТРАФ И ВОЗМЕЩЕНИЕ
// -----------------------------------------------------

// loss_resolution имеет UNIQUE loan_id.
//
// Следовательно, для одного займа невозможно
// одновременно зарегистрировать штраф и возмещение.
//
// Перед созданием loss_resolution:
//
// - loan.lost_at должен быть заполнен.
//
// Для resolution_type = fine:
//
// - просрочка должна превышать один год;
// - edition_price_at_assessment копируется
//   из book_edition.catalog_price;
// - fine_amount = 3 * edition_price_at_assessment;
// - fine_assessed_at фиксирует начисление;
// - штраф считается оплаченным в тот же момент;
// - экземпляр остается утерянным.
//
// Для resolution_type = replacement:
//
// - дата подачи заявления не должна быть позже
//   due_at + 1 год;
// - replacement_copy_id должен относиться
//   к тому же edition_id, что и потерянная книга;
// - replacement_copy_id не должен совпадать
//   с original_copy_id;
// - экземпляр должен быть доступен;
// - экземпляр не должен быть списан или утерян;
// - один экземпляр нельзя использовать
//   для нескольких возмещений.
//
// Составной FK обеспечивает совпадение edition_id.
//
// UNIQUE(replacement_copy_id) исключает повторное
// использование экземпляра.
//
// Проверка доступности экземпляра выполняется триггером.
//
// Заявление считается исполненным только при заполнении
// replacement_copy_id и replacement_received_at.
//
// До исполнения заявления задолженность сохраняется.
//
// -----------------------------------------------------
// 7. СПИСАНИЕ
// -----------------------------------------------------

// Составной FK:
//
// inventory_writeoff.(loan_id, copy_id)
//     -> loan.(loan_id, copy_id)
//
// гарантирует, что указанная выдача действительно
// относится к списываемому физическому экземпляру.
//
// Если loan_id указан, триггер дополнительно проверяет:
//
// - соответствие экземпляра;
// - отсутствие активной выдачи;
// - наличие loan.lost_at для списания невозвращенной книги.
//
// Списание не изменяет loan и loss_resolution.
//
// Следовательно, списание книги не аннулирует
// задолженность читателя.
//
// Если книга списана до возмещения, обязательство
// остается незакрытым.
//
// -----------------------------------------------------
// 8. ИСТОРИЯ АКТИВНОСТИ ЧИТАТЕЛЕЙ
// -----------------------------------------------------

// Отдельная таблица member_activity отсутствует.
//
// Последняя активность определяется по максимальной дате
// следующих событий:
//
// - library_member.registered_at;
// - member_visit.visited_at;
// - loan.issued_at;
// - loan.returned_at;
// - loan_renewal.renewed_at;
// - loss_resolution.application_submitted_at;
// - loss_resolution.replacement_received_at.
//
// Автоматически начисленный штраф не считается
// пользовательской активностью.
//
// Посещения учитываются отдельно в member_visit.
//
// Это исключает зависимость учета активности
// от ручного дублирования событий.
//
// При отчислении триггер проверяет:
//
// 1. Последний статус читателя — active.
// 2. Отсутствие активных выдач.
// 3. Отсутствие неурегулированных потерь.
// 4. Отсутствие невыполненных заявлений на возмещение.
// 5. Отсутствие активности более одного года.
//
// При создании нового читателя автоматически добавляется
// первая запись member_status_history со статусом active.
//
// Текущий статус определяется последней записью истории.
// Изменение статуса выполняется только добавлением
// новой записи, без изменения предыдущих.
//
// -----------------------------------------------------
// 9. ИСТОРИЯ ХАРАКТЕРИСТИК КНИГ
// -----------------------------------------------------

// При выдаче триггер автоматически фиксирует:
//
// - book_id_at_issue;
// - book_title_at_issue;
// - genre_id_at_issue;
// - genre_name_at_issue.
//
// Дополнительно создаются строки loan_author
// для всех авторов из book_author.
//
// loan_author сохраняет:
//
// - author_id;
// - author_name_at_issue;
// - author_order_at_issue.
//
// Изменение жанра, названия или авторского состава
// в текущем каталоге не изменяет исторические выдачи.
//
// Исторические отчеты используют snapshot-поля,
// а не текущие значения book и book_author.
//
// -----------------------------------------------------
// 10. ИСТОРИЯ УВЕДОМЛЕНИЙ
// -----------------------------------------------------

// У одного loan может быть любое количество overdue_notice.
//
// При формировании уведомления проверяется:
//
// generated_at > loan.due_at + INTERVAL '1 month'.
//
// Каждое уведомление сохраняется отдельно.
//
// Исправление ранее сформированного документа
// оформляется новой записью, а не перезаписью старой.
//
// -----------------------------------------------------
// 11. ПРОВЕРКИ ДАТ И ДЕНЕЖНЫХ СУММ
// -----------------------------------------------------

// Через CHECK обеспечиваются:
//
// - положительное количество в acquisition_item;
// - неотрицательная первоначальная цена экземпляра;
// - неотрицательная каталожная цена издания;
// - дата возврата не раньше выдачи;
// - дата потери не раньше выдачи;
// - новая дата возврата позже предыдущей;
// - номер продления от 1 до 3;
// - неотрицательная стоимость штрафа;
// - штраф равен тройной стоимости издания.
//
// Через триггеры обеспечиваются:
//
// - неотрицательная итоговая стоимость после корректировок;
// - неотрицательное итоговое количество экземпляров;
// - согласованность дат с историей операций;
// - соответствие фактического количества экземпляров
//   количеству в проведенных документах.
//
// -----------------------------------------------------
// 12. ОТЧЕТЫ
// -----------------------------------------------------

// Наиболее популярные книги за произвольный период:
//
// COUNT(loan.loan_id)
// GROUP BY loan.book_id_at_issue
// WHERE issued_at BETWEEN period_start AND period_end.
//
// Для отображения названия используется book_title_at_issue.
//
// Наиболее популярные авторы за произвольный период:
//
// COUNT(loan_author.loan_id)
// GROUP BY loan_author.author_id
// JOIN loan ON loan_author.loan_id = loan.loan_id
// WHERE loan.issued_at BETWEEN period_start AND period_end.
//
// Каждая выдача соавторской книги учитывается
// отдельно для каждого автора.
//
// Количество прочитанных книг на человека за год:
//
// COUNT(loan.loan_id)
// GROUP BY loan.member_id
// WHERE returned_at BETWEEN year_start AND year_end.
//
// Учитываются только фактически возвращенные книги.
// Продления не увеличивают количество прочитанных книг.
//
// Жанровые предпочтения за год:
//
// COUNT(loan.loan_id)
// GROUP BY genre_id_at_issue, genre_name_at_issue
// WHERE issued_at BETWEEN year_start AND year_end.
//
// Используется жанр на момент выдачи.
//
// Финансовый отчет за год:
//
// Основные фонды:
//
// SUM(acquisition_item.quantity * acquisition_item.unit_price)
// для проведенных документов acquisition.
//
// Корректировки:
//
// SUM(acquisition_adjustment.amount_delta)
// по году posted_at.
//
// Штрафы:
//
// SUM(loss_resolution.fine_amount)
// для resolution_type = fine,
// по году fine_assessed_at.
//
// Бесплатные поступления учитываются с нулевой стоимостью,
// если им не назначена отдельная учетная оценка.
//
// Первоначальный фонд учитывается через acquisition_type
// opening_balance.
```

**Ключевые изменения относительно предыдущей версии:**

- Удалены дублирующие статусы `book_copy.status`, `loan.status` и `library_member.status`. Состояния теперь определяются по операционным записям и истории.
- Добавлена `active_loan`: три слота на читателя обеспечивают ограничение выдачи, а уникальность экземпляра защищает от двойной выдачи.
- Добавлена `acquisition_adjustment` с историей изменений количества и стоимости. Первоначальные данные проведенных документов сохраняются.
- Добавлена отмена ошибочной регистрации экземпляра без физического удаления записи.
- Штраф и возмещение объединены в `loss_resolution`, поэтому их одновременное оформление невозможно.
- Добавлены исторические снимки жанра, названия и авторов непосредственно при выдаче.
- Уведомления теперь допускают несколько записей на один займ.
- Удалена дублирующая таблица активности: посещения регистрируются отдельно, остальные события берутся из операционных таблиц.

Схема сохраняет возможность формирования всех пяти отчетов и реализации операций исходного задания. При этом для полноценной работы ограничений потребуется реализовать перечисленные в комментариях триггеры PostgreSQL и выполнять связанные изменения в одной транзакции.

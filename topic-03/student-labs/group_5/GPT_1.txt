
Project auto_service {
  database_type: 'PostgreSQL'
  Note: 'База данных станции по ремонту автомобилей'
}

// 1. Модели автомобилей
Table car_models {
  model_id int [pk, increment]
  brand varchar(100) [not null]
  model_name varchar(100) [not null]
  generation varchar(100)
  production_year_from int
  production_year_to int
  description text

  indexes {
    (brand, model_name, generation) [unique]
  }
}

// 2. Клиенты
Table clients {
  client_id int [pk, increment]
  last_name varchar(100) [not null]
  first_name varchar(100) [not null]
  middle_name varchar(100)
  phone varchar(30) [not null]
  email varchar(150)
  address text
  registration_date date [not null]
  status varchar(20) [not null, default: 'active']

  indexes {
    phone
    email
  }
}

// 3. Автомобили
Table cars {
  car_id int [pk, increment]
  model_id int [not null, ref: > car_models.model_id]
  vin varchar(17) [unique]
  registration_number varchar(20)
  manufacture_year int
  color varchar(50)
  status varchar(20) [not null, default: 'active']

  indexes {
    registration_number
    model_id
  }
}

// 4. История владения автомобилями
Table car_ownership {
  ownership_id int [pk, increment]
  car_id int [not null, ref: > cars.car_id]
  client_id int [not null, ref: > clients.client_id]
  ownership_start date [not null]
  ownership_end date
  is_current boolean [not null, default: true]

  indexes {
    (car_id, ownership_start)
    (client_id, is_current)
  }
}

// 5. Склад запчастей и расходных материалов
Table spare_parts {
  part_id int [pk, increment]
  part_name varchar(150) [not null]
  part_category varchar(100)
  manufacturer varchar(100)
  part_number varchar(100)
  condition_type varchar(10) [not null, note: 'new или used']
  purchase_price decimal(12,2)
  selling_price decimal(12,2) [not null]
  quantity int [not null, default: 0]
  unit varchar(30) [not null, default: 'шт.']
  description text
  is_active boolean [not null, default: true]

  indexes {
    part_number
    part_name
  }
}

// 6. Совместимость запчастей с моделями
Table part_model_compatibility {
  compatibility_id int [pk, increment]
  part_id int [not null, ref: > spare_parts.part_id]
  model_id int [not null, ref: > car_models.model_id]

  indexes {
    (part_id, model_id) [unique]
  }
}

// 7. Справочник работ
Table services {
  service_id int [pk, increment]
  service_name varchar(150) [not null]
  description text
  service_price decimal(12,2) [not null]
  estimated_duration_minutes int
  is_periodic boolean [not null, default: false]
  is_active boolean [not null, default: true]
}

// 8. Периодичность работ
Table service_intervals {
  interval_id int [pk, increment]
  service_id int [not null, unique, ref: - services.service_id]
  interval_km int
  interval_months int
}

// 9. Совместимость работ с моделями
Table service_model_compatibility {
  compatibility_id int [pk, increment]
  service_id int [not null, ref: > services.service_id]
  model_id int [not null, ref: > car_models.model_id]

  indexes {
    (service_id, model_id) [unique]
  }
}

// 10. Заказы
Table orders {
  order_id int [pk, increment]
  car_id int [not null, ref: > cars.car_id]
  client_id int [not null, ref: > clients.client_id]
  order_date datetime [not null]
  completion_date datetime
  mileage int [not null]
  status varchar(20) [not null, default: 'new']
  total_parts_cost decimal(12,2) [not null, default: 0]
  total_services_cost decimal(12,2) [not null, default: 0]
  total_cost decimal(12,2) [not null, default: 0]
  description text

  indexes {
    car_id
    client_id
    order_date
    status
  }
}

// 11. Запчасти в заказах
Table order_parts {
  order_part_id int [pk, increment]
  order_id int [not null, ref: > orders.order_id]
  part_id int [not null, ref: > spare_parts.part_id]
  quantity int [not null]
  unit_price decimal(12,2) [not null]
  total_price decimal(12,2) [not null]

  indexes {
    order_id
    part_id
  }
}

// 12. Работы в заказах
Table order_services {
  order_service_id int [pk, increment]
  order_id int [not null, ref: > orders.order_id]
  service_id int [not null, ref: > services.service_id]
  quantity int [not null, default: 1]
  unit_price decimal(12,2) [not null]
  total_price decimal(12,2) [not null]

  indexes {
    order_id
    service_id
  }
}

// 13. История выполнения периодических работ
Table car_service_history {
  history_id int [pk, increment]
  car_id int [not null, ref: > cars.car_id]
  service_id int [not null, ref: > services.service_id]
  order_service_id int [not null, unique, ref: - order_services.order_service_id]
  execution_date datetime [not null]
  mileage int [not null]

  indexes {
    (car_id, service_id, execution_date)
  }
}

// 14. Напоминания о периодическом обслуживании
Table service_reminders {
  reminder_id int [pk, increment]
  car_id int [not null, ref: > cars.car_id]
  client_id int [not null, ref: > clients.client_id]
  service_id int [not null, ref: > services.service_id]
  reminder_date date [not null]
  due_date date
  due_mileage int
  status varchar(20) [not null, default: 'pending']
  created_at datetime [not null]
  completed_at datetime

  indexes {
    (car_id, service_id, status)
    client_id
    reminder_date
  }
}

// 15. Печатные письма клиентам
Table reminder_letters {
  letter_id int [pk, increment]
  client_id int [not null, ref: > clients.client_id]
  reminder_id int [not null, ref: > service_reminders.reminder_id]
  creation_date datetime [not null]
  letter_text text [not null]
  delivery_method varchar(20) [not null, default: 'print']
  status varchar(20) [not null, default: 'created']

  indexes {
    reminder_id
    client_id
  }
}

// 16. Снятие с учета
Table deregistrations {
  deregistration_id int [pk, increment]
  client_id int [ref: > clients.client_id]
  car_id int [ref: > cars.car_id]
  deregistration_date datetime [not null]
  reason text
}

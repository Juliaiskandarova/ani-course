// База данных: Альпинистский клуб
//
// ОСНОВНЫЕ ДОПУЩЕНИЯ:
//
// 1. Каждая группа формируется для одного конкретного похода
//    по одному маршруту к одной вершине.
//    Для повторного похода создается новая группа.
//
// 2. Один маршрут ведет к одной вершине.
//    У вершины может быть несколько маршрутов.
//
// 3. У маршрута может быть несколько стартовых и несколько
//    конечных населенных пунктов.
//    Это допустимые точки начала и окончания маршрута,
//    а не последовательные остановки.
//
// 4. Время прохождения маршрута хранится в минутах.
//    Ожидаемое время выхода — планируемое время завершения
//    маршрута, а не время начала похода.
//
// 5. Группа имеет одного старшего, который обязательно
//    входит в ее состав.
//
// 6. Участники группы фиксируются для данного похода.
//    Исторический состав не изменяется после завершения похода.
//
// 7. Успешное покорение вершины фиксируется отдельной записью.
//    Вершина считается покоренной группой, если хотя бы один
//    участник группы достиг вершины.
//    При этом отдельно фиксируются альпинисты,
//    действительно достигшие вершины.
//
// 8. Происшествия на маршруте учитываются через эвакуации.
//    Для одной группы допускается несколько эвакуационных событий.
//    В каждой эвакуации хранится список спасенных участников.
//
// 9. Самые опасные горы (топ-5) определяются по доле походов,
//    в которых происходила хотя бы одна эвакуация,
//    среди всех фактически начатых походов к данной вершине.
//    При равенстве долей выше располагается гора
//    с большим абсолютным числом таких походов.
//    Горы без начатых походов в рейтинг не включаются.
//
// 10. Самые крутые альпинисты определяются по количеству
//     успешных личных восхождений.
//     Дополнительно доступны число уникальных покоренных вершин,
//     высота вершин и сложность пройденных маршрутов.
//     Рейтинг строится по всей истории, включая неактивных.
//
// 11. Статусы, допустимые диапазоны значений, временные
//     ограничения и правила изменения состояний контролируются
//     бизнес-логикой или ограничениями БД, не выраженными в DBML.


// ============================================================
// ПЕРЕЧИСЛЕНИЯ
// ============================================================

Enum climber_status {
  active
  departed
  deceased
}

Enum group_status {
  forming
  scheduled
  on_route
  completed
  evacuated
  cancelled
}

Enum endpoint_role {
  start
  finish
}


// ============================================================
// АЛЬПИНИСТЫ
// ============================================================

Table climbers {
  certificate_no varchar(50) [pk, not null, note: 'Уникальный номер сертификата альпиниста']
  full_name varchar(255) [not null]
  status climber_status [not null, default: 'active']
  registered_at timestamp [not null]
  status_changed_at timestamp
  notes text

  Note: '''
    Реестр альпинистов.

    active — состоит в клубе и может участвовать в новых походах.
    departed — покинул клуб.
    deceased — погиб.

    При уходе или гибели запись не удаляется.
    История участия и восхождений сохраняется.

    В новые группы допускаются только активные альпинисты.
    Изменение статуса не изменяет исторические данные.
  '''
}


// ============================================================
// ГОРНЫЕ ВЕРШИНЫ
// ============================================================

Table peaks {
  id int [pk, increment]
  country varchar(100) [not null]
  mountain_name varchar(255) [not null]
  elevation_m int [not null, note: 'Высота вершины в метрах, положительное число']
  description text

  indexes {
    (country, mountain_name)
  }

  Note: '''
    Каталог горных вершин.

    Название не является уникальным идентификатором:
    в одной стране могут существовать одноименные вершины.

    Одна вершина может иметь множество маршрутов.
  '''
}


// ============================================================
// НАСЕЛЕННЫЕ ПУНКТЫ
// ============================================================

Table settlements {
  id int [pk, increment]
  name varchar(255) [not null]
  country varchar(100) [not null]
  region varchar(255)

  indexes {
    (country, name)
  }

  Note: '''
    Справочник населенных пунктов.

    Один населенный пункт может использоваться
    в нескольких маршрутах.

    Регион позволяет различать одноименные пункты.
  '''
}


// ============================================================
// МАРШРУТЫ
// ============================================================

Table routes {
  id int [pk, increment]
  peak_id int [not null]
  route_name varchar(255) [not null]
  duration_minutes int [not null, note: 'Плановая продолжительность, больше нуля']
  difficulty_category varchar(10) [not null, note: 'Категория сложности, например 1A, 1B, 2A ... 6B']
  description text

  indexes {
    peak_id
    difficulty_category
  }

  Note: '''
    Маршрут предназначен для восхождения на одну вершину.

    Категория сложности задается по принятой клубом шкале.
    Допустимые значения проверяются бизнес-логикой.

    Для каждого маршрута должен существовать
    хотя бы один стартовый и один конечный населенный пункт.

    Населенные пункты маршрута хранятся отдельно,
    поскольку стартов и финишей может быть несколько.
  '''
}


// ============================================================
// НАЧАЛЬНЫЕ И КОНЕЧНЫЕ ПУНКТЫ МАРШРУТОВ
// ============================================================

Table route_endpoints {
  route_id int [not null]
  settlement_id int [not null]
  role endpoint_role [not null]

  indexes {
    (route_id, settlement_id, role) [pk]
    settlement_id
  }

  Note: '''
    Связь маршрутов с населенными пунктами M:N.

    start — возможный стартовый населенный пункт.
    finish — возможный конечный населенный пункт.

    Для одного маршрута допускается несколько записей
    каждого типа.

    Обязательное наличие старта и финиша для маршрута
    проверяется при завершении его создания.
  '''
}


// ============================================================
// АЛЬПИНИСТСКИЕ ГРУППЫ И ПОХОДЫ
// ============================================================

Table climbing_groups {
  id int [pk, increment]
  group_name varchar(255) [not null]

  route_id int [not null]
  leader_certificate_no varchar(50)

  status group_status [not null, default: 'forming']

  created_at timestamp [not null]

  planned_start_at timestamp [not null]
  expected_exit_at timestamp [not null]

  actual_start_at timestamp
  actual_exit_at timestamp

  notes text

  indexes {
    route_id
    leader_certificate_no
    (status, expected_exit_at)
  }

  Note: '''
    Каждая запись представляет группу, сформированную
    для конкретного похода.

    Целевая вершина определяется через маршрут:
    climbing_groups -> routes -> peaks.

    Статусы:
    forming — состав формируется;
    scheduled — группа сформирована, поход запланирован;
    on_route — группа находится на маршруте;
    completed — поход завершен;
    evacuated — поход завершен эвакуацией;
    cancelled — поход отменен.

    Старший должен быть участником той же группы.
    Это обеспечивается составным внешним ключом.

    Во время формирования группы старший может быть NULL,
    чтобы сначала создать группу и добавить участников.
    Перед переводом в scheduled старший обязателен.

    planned_start_at — плановое начало похода.
    expected_exit_at — ожидаемое время выхода с маршрута.
    actual_start_at — фактическое начало похода.
    actual_exit_at — фактическое завершение похода.

    Для статуса on_route требуется:
    actual_start_at IS NOT NULL,
    actual_exit_at IS NULL.

    expected_exit_at должен быть позже planned_start_at.

    Одновременное участие альпиниста в пересекающихся
    по времени активных походах запрещается бизнес-логикой.

    Завершение маршрута не означает покорения вершины:
    успешное восхождение учитывается отдельно.
  '''
}


// ============================================================
// СОСТАВ ГРУПП
// ============================================================

Table group_members {
  group_id int [not null]
  climber_certificate_no varchar(50) [not null]
  added_at timestamp [not null]

  indexes {
    (group_id, climber_certificate_no) [pk]
    climber_certificate_no
  }

  Note: '''
    Состав группы: связь групп и альпинистов M:N.

    Один альпинист может участвовать в разных походах.
    В одной группе альпинист учитывается только один раз.

    В момент добавления альпинист должен быть активным.

    После начала похода состав группы считается
    исторически зафиксированным.

    Назначенный старший обязательно присутствует
    в этой таблице для соответствующей группы.
  '''
}


// ============================================================
// УСПЕШНЫЕ ВОСХОЖДЕНИЯ ГРУПП
// ============================================================

Table summit_ascents {
  id int [pk, increment]
  group_id int [not null, unique]
  reached_at timestamp [not null, note: 'Дата и время достижения вершины']
  notes text

  indexes {
    (id, group_id) [unique]
  }

  Note: '''
    Факт достижения целевой вершины группой.

    Одна группа совершает не более одного
    зарегистрированного успешного восхождения
    в рамках данного похода.

    Вершина определяется через маршрут группы.

    Отсутствие записи для завершенного похода
    означает, что успешное восхождение не зафиксировано.

    Для успешного восхождения должен существовать
    хотя бы один участник в summit_participants.

    Дата достижения вершины не может предшествовать
    фактическому началу похода или наступать
    после его фактического завершения.

    Таблица позволяет считать количество успешных
    восхождений по каждой вершине, группе и маршруту.
  '''
}


// ============================================================
// АЛЬПИНИСТЫ, ДОСТИГШИЕ ВЕРШИНЫ
// ============================================================

Table summit_participants {
  ascent_id int [not null]
  group_id int [not null]
  climber_certificate_no varchar(50) [not null]

  indexes {
    (ascent_id, climber_certificate_no) [pk]
    (group_id, climber_certificate_no)
  }

  Note: '''
    Участники, лично достигшие вершины.

    Факт членства в группе сам по себе не означает,
    что альпинист покорил вершину.

    Составные внешние ключи гарантируют:
    1. Восхождение принадлежит указанной группе.
    2. Альпинист входит в состав указанной группы.

    История альпиниста получается через:
    summit_participants -> summit_ascents
    -> climbing_groups -> routes -> peaks.

    Для рейтинга альпинистов можно вычислять:
    - общее количество личных успешных восхождений;
    - количество уникальных покоренных вершин;
    - максимальную высоту покоренной вершины;
    - категории сложности пройденных маршрутов.
  '''
}


// ============================================================
// ЭВАКУАЦИИ
// ============================================================

Table evacuations {
  id int [pk, increment]
  group_id int [not null]
  occurred_at timestamp [not null, note: 'Дата и время эвакуации']
  reason text

  indexes {
    group_id
    occurred_at
    (id, group_id) [unique]
  }

  Note: '''
    Эвакуационное событие, произошедшее на маршруте.

    Каждая эвакуация относится к конкретной группе.
    В одной группе возможно несколько событий.

    Причина может содержать описание происшествия:
    непогода, травма, лавина и т.д.

    Эвакуация должна происходить не ранее
    фактического начала похода.

    Список спасенных хранится в evacuation_rescued.

    Для оценки опасности вершины подсчитываются
    уникальные группы, у которых есть эвакуации,
    а не количество эвакуационных событий.

    Для завершенного эвакуацией похода состояние группы
    переводится в evacuated и фиксируется actual_exit_at.
  '''
}


// ============================================================
// СПАСЕННЫЕ ПРИ ЭВАКУАЦИИ
// ============================================================

Table evacuation_rescued {
  evacuation_id int [not null]
  group_id int [not null]
  climber_certificate_no varchar(50) [not null]

  indexes {
    (evacuation_id, climber_certificate_no) [pk]
    (group_id, climber_certificate_no)
  }

  Note: '''
    Список спасенных альпинистов для каждой эвакуации.

    Один альпинист не может быть записан дважды
    в список спасенных одной эвакуации.

    Составные внешние ключи гарантируют:
    1. Указанная группа соответствует эвакуации.
    2. Спасенный альпинист входит в эту группу.

    Отсутствие альпиниста в списке спасенных
    не означает автоматически его гибель.

    Статус deceased устанавливается отдельно
    в реестре альпинистов.
  '''
}


// ============================================================
// ВНЕШНИЕ КЛЮЧИ И СВЯЗИ
// ============================================================

// Одна вершина — много маршрутов
Ref: routes.peak_id > peaks.id

// Маршруты — населенные пункты (M:N)
Ref: route_endpoints.route_id > routes.id
Ref: route_endpoints.settlement_id > settlements.id

// Один маршрут — много групповых походов
Ref: climbing_groups.route_id > routes.id

// Группы — альпинисты (M:N)
Ref: group_members.group_id > climbing_groups.id
Ref: group_members.climber_certificate_no > climbers.certificate_no

// Старший группы должен входить в состав этой же группы.
// Составной FK допускает NULL при формировании группы.
Ref: climbing_groups.(id, leader_certificate_no) > group_members.(group_id, climber_certificate_no)

// Одна группа — не более одного успешного восхождения
Ref: summit_ascents.group_id > climbing_groups.id

// Участники успешного восхождения
Ref: summit_participants.(ascent_id, group_id) > summit_ascents.(id, group_id)
Ref: summit_participants.(group_id, climber_certificate_no) > group_members.(group_id, climber_certificate_no)

// Одна группа — множество возможных эвакуаций
Ref: evacuations.group_id > climbing_groups.id

// Список спасенных при каждой эвакуации
Ref: evacuation_rescued.(evacuation_id, group_id) > evacuations.(id, group_id)
Ref: evacuation_rescued.(group_id, climber_certificate_no) > group_members.(group_id, climber_certificate_no)


// ============================================================
// РЕАЛИЗУЕМОСТЬ ОПЕРАЦИЙ И ОТЧЕТОВ
// ============================================================

// ОПЕРАЦИИ:
//
// Новая вершина:
//   peaks
//
// Новый маршрут:
//   routes + route_endpoints + settlements
//
// Новый альпинист:
//   climbers
//
// Перевод в неактивное состояние:
//   climbers.status = departed / deceased
//
// Формирование группы:
//   climbing_groups + group_members
//
// Назначение старшего:
//   climbing_groups.leader_certificate_no
//
//
// ОТЧЕТ 1. АКТИВНЫЕ ГРУППЫ НА МАРШРУТЕ
//
//   climbing_groups (status = on_route)
//   -> routes
//   -> group_members
//   -> climbers
//
//   Старший определяется по leader_certificate_no.
//   Состав — по group_members.
//   Ожидаемый выход — expected_exit_at.
//   Группировка — по climbing_groups.id.
//
//
// ОТЧЕТ 2. ИСТОРИЯ ПОКОРЕНИЯ ВЕРШИН АЛЬПИНИСТОМ
//
//   climbers
//   -> summit_participants
//   -> summit_ascents
//   -> climbing_groups
//   -> routes
//   -> peaks
//
//   Отображаются дата восхождения, вершина,
//   высота, маршрут, категория сложности и группа.
//
//
// ОТЧЕТ 3. САМЫЕ ОПАСНЫЕ ГОРЫ (ТОП-5)
//
//   peaks
//   -> routes
//   -> climbing_groups
//   -> evacuations
//
//   Для каждой вершины вычисляются:
//
//   started_groups = количество уникальных групп,
//     где actual_start_at IS NOT NULL.
//
//   evacuated_groups = количество уникальных групп,
//     где actual_start_at IS NOT NULL
//     и имеется хотя бы одна запись в evacuations.
//
//   danger_rate = evacuated_groups / started_groups.
//
//   Результат сортируется по danger_rate DESC,
//   evacuated_groups DESC, затем ограничивается топ-5.
//
//
// ОТЧЕТ 4. САМЫЕ КРУТЫЕ АЛЬПИНИСТЫ
//
//   climbers
//   -> summit_participants
//   -> summit_ascents
//   -> climbing_groups
//   -> routes
//   -> peaks
//
//   Основной критерий — количество личных восхождений.
//
//   Дополнительные показатели:
//   - число различных покоренных вершин;
//   - максимальная высота покоренной вершины;
//   - суммарная высота покоренных вершин;
//   - сложность маршрутов.
//
//   Сортировка по числу восхождений DESC,
//   затем по числу уникальных вершин DESC,
//   затем по максимальной высоте DESC.
//
//   Можно вывести полный рейтинг или ограничить топ-N.
//   Альпинисты без восхождений имеют нулевые показатели.

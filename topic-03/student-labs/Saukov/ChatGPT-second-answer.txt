// ============================================================
// АЛЬПИНИСТСКИЙ КЛУБ — ИСПРАВЛЕННАЯ СХЕМА
// ============================================================
//
// ОСНОВНЫЕ ДОПУЩЕНИЯ
//
// 1. Одна группа соответствует одному конкретному походу
//    по одному маршруту к одной вершине.
//    Для повторного похода создается новая группа.
//
// 2. Один маршрут относится к одной вершине.
//    У вершины может быть несколько маршрутов.
//
// 3. Маршрут допускает несколько стартовых и несколько
//    конечных населенных пунктов. Они рассматриваются
//    как допустимые альтернативы начала и окончания маршрута.
//
// 4. Продолжительность маршрута хранится в минутах.
//    expected_exit_at означает планируемое время завершения
//    похода, а не время достижения вершины.
//
// 5. Старший группы обязательно входит в ее состав.
//    При формировании группы старший может быть еще не назначен.
//
// 6. Исходный состав группы сохраняется для истории.
//    После фактического начала похода состав не изменяется.
//
// 7. Участие в походе не означает покорения вершины.
//    Успешное восхождение группы и личное достижение вершины
//    каждым альпинистом учитываются отдельно.
//
// 8. Одна группа может иметь не более одного зарегистрированного
//    успешного восхождения в рамках конкретного похода.
//
// 9. Участник, эвакуированный из похода, в этот же поход
//    на маршрут не возвращается.
//
// 10. Частичная эвакуация не завершает поход всей группы.
//     Если оставшиеся участники продолжают маршрут,
//     статус группы сохраняется как on_route.
//
// 11. После начала похода его маршрут, исходный состав
//     и исторически значимые характеристики маршрута
//     должны быть защищены от изменения.
//
// 12. CHECK обеспечивает только ограничения одной строки.
//     Проверки между таблицами, истории изменений и переходов
//     состояний выполняются триггерами или бизнес-логикой.
//
// 13. Удаление исторических записей запрещается правилами
//     приложения и правами доступа. Каскадное удаление
//     для исторических связей не используется.
//
// 14. Время хранится в timestamp.
//     Все временные значения трактуются в единой принятой
//     клубом временной системе (например, UTC).
//
//
// ИЗМЕНЕНИЯ ПО СРАВНЕНИЮ С ПЕРВОЙ ВЕРСИЕЙ
//
// - Связь climbing_groups — summit_ascents исправлена на 1:1.
// - Уникальность summit_ascents.group_id сохранена.
// - Добавлены CHECK для числовых значений и временных интервалов.
// - Добавлены CHECK для согласованности статуса и дат похода.
// - Плановые даты допускают NULL только при формировании
//   или отмене похода.
// - Уточнены правила частичной эвакуации.
// - Добавлена уникальность спасения участника в рамках группы.
// - Уточнена защита исторических маршрутов и вершин.
// - Сохранены составные внешние ключи принадлежности группе.


// ============================================================
// ПЕРЕЧИСЛЕНИЯ
// ============================================================

Enum climber_status {
  active
  departed
  deceased
}

Enum group_status {
  forming
  scheduled
  on_route
  completed
  evacuated
  cancelled
}

Enum endpoint_role {
  start
  finish
}

// Допущение о шкале сложности:
// клуб использует категории от 1A до 6B.
// Если применяется другая классификация, Enum необходимо
// изменить согласованно с правилами клуба.
//
// Порядок категорий в перечне соответствует
// возрастанию сложности.

Enum route_difficulty {
  "1A"
  "1B"
  "2A"
  "2B"
  "3A"
  "3B"
  "4A"
  "4B"
  "5A"
  "5B"
  "6A"
  "6B"
}


// ============================================================
// 1. АЛЬПИНИСТЫ
// ============================================================

Table climbers {
  certificate_no varchar(50) [pk, not null]
  full_name varchar(255) [not null]

  status climber_status [not null, default: 'active']

  registered_at timestamp [not null]
  status_changed_at timestamp

  notes text

  indexes {
    status
    full_name
  }

  checks {
    `status = 'active' OR status_changed_at IS NOT NULL` [name: 'chk_climber_inactive_date']

    `status_changed_at IS NULL OR status_changed_at >= registered_at` [name: 'chk_climber_status_date']
  }

  Note: '''
    Учет альпинистов по уникальному номеру сертификата.

    active — активный участник клуба.
    departed — покинул клуб.
    deceased — погиб.

    При уходе или гибели запись не удаляется.

    Для добавления в новую группу альпинист должен
    иметь статус active.

    Это правило требует проверки при добавлении
    в group_members.

    Изменение статуса не удаляет альпиниста
    из истории ранее состоявшихся походов.

    Переход из deceased обратно в active запрещается
    бизнес-логикой или триггером.

    status_changed_at фиксирует последнее изменение
    статуса; для нового active допускается NULL.
  '''
}


// ============================================================
// 2. ГОРНЫЕ ВЕРШИНЫ
// ============================================================

Table peaks {
  id int [pk, increment]

  country varchar(100) [not null]
  mountain_name varchar(255) [not null]

  elevation_m int [not null, check: `elevation_m > 0`]

  description text

  indexes {
    (country, mountain_name)
  }

  Note: '''
    Каталог горных вершин.

    Высота обязательно положительная.

    Название горы не считается уникальным:
    возможны одноименные вершины.

    Одна вершина может иметь множество маршрутов.

    ЗАЩИТА ИСТОРИИ:

    Если существует фактически начатый поход
    к вершине, изменение country, mountain_name
    и elevation_m должно быть ограничено.

    Эти поля используются в исторических отчетах
    и рейтингах.

    Запрет изменения и удаления обеспечивается
    триггером или бизнес-логикой.

    CHECK и FK сами по себе этот запрет не обеспечивают.

    Существенные изменения идентификации вершины
    оформляются отдельной записью.

    Исправление достоверной ошибки в справочнике
    допускается только контролируемой операцией
    с проверкой влияния на исторические данные.
  '''
}


// ============================================================
// 3. НАСЕЛЕННЫЕ ПУНКТЫ
// ============================================================

Table settlements {
  id int [pk, increment]

  name varchar(255) [not null]
  country varchar(100) [not null]
  region varchar(255)

  indexes {
    (country, name)
  }

  Note: '''
    Справочник населенных пунктов.

    Один населенный пункт может быть связан
    с несколькими маршрутами.

    region позволяет различать одноименные пункты.

    Исторически использованные населенные пункты
    не должны удаляться или произвольно переименовываться.

    Такие изменения контролируются бизнес-логикой.
  '''
}


// ============================================================
// 4. МАРШРУТЫ
// ============================================================

Table routes {
  id int [pk, increment]

  peak_id int [not null]

  route_name varchar(255) [not null]

  duration_minutes int [not null, check: `duration_minutes > 0`]

  difficulty_category route_difficulty [not null]

  description text

  indexes {
    peak_id
    difficulty_category
  }

  Note: '''
    Маршрут восхождения на конкретную вершину.

    Продолжительность строго положительная.

    Категория сложности ограничена Enum route_difficulty.

    Для каждого маршрута обязателен хотя бы один
    стартовый и один конечный населенный пункт.

    Наличие таких пунктов проверяется бизнес-логикой
    или триггером: обычный CHECK не может проверять
    количество строк в route_endpoints.

    ЗАЩИТА ИСТОРИИ:

    Если маршрут использовался хотя бы одной группой,
    у которой был фактически начат поход, запрещается
    изменять:

    - peak_id;
    - route_name;
    - duration_minutes;
    - difficulty_category;
    - исторически значимое описание маршрута;
    - набор стартовых и конечных населенных пунктов.

    Для нового варианта маршрута создается
    новая запись routes.

    Проверка наличия начатых походов выполняется
    триггером или бизнес-логикой.

    Внешний ключ peak_id обеспечивает существование
    вершины, но не запрещает переназначение маршрута
    другой существующей вершине.

    Поэтому FK сам по себе не защищает историю.
  '''
}


// ============================================================
// 5. НАЧАЛЬНЫЕ И КОНЕЧНЫЕ ПУНКТЫ МАРШРУТА
// ============================================================

Table route_endpoints {
  route_id int [not null]

  settlement_id int [not null]

  role endpoint_role [not null]

  indexes {
    (route_id, settlement_id, role) [pk]
    settlement_id
  }

  Note: '''
    Связь маршрутов с населенными пунктами M:N.

    start — допустимый старт маршрута.
    finish — допустимое окончание маршрута.

    Один населенный пункт может быть одновременно
    стартовым и конечным для одного маршрута.

    Для каждого маршрута необходимы:
    - минимум одна запись start;
    - минимум одна запись finish.

    Эти минимальные количества контролируются
    триггером или бизнес-логикой.

    Добавление, изменение или удаление точек
    маршрута, уже использованного в начатом походе,
    запрещается триггером или бизнес-логикой.

    Первичный ключ запрещает дублирование
    одинаковой комбинации маршрут/пункт/роль.
  '''
}


// ============================================================
// 6. АЛЬПИНИСТСКИЕ ГРУППЫ И ПОХОДЫ
// ============================================================

Table climbing_groups {
  id int [pk, increment]

  group_name varchar(255) [not null]

  route_id int [not null]

  leader_certificate_no varchar(50)

  status group_status [not null, default: 'forming']

  created_at timestamp [not null]

  planned_start_at timestamp
  expected_exit_at timestamp

  actual_start_at timestamp
  actual_exit_at timestamp

  notes text

  indexes {
    route_id
    leader_certificate_no
    (status, expected_exit_at)
    actual_start_at
  }

  checks {

    // Плановые даты либо обе отсутствуют,
    // либо образуют корректный интервал.

    `(planned_start_at IS NULL AND expected_exit_at IS NULL) OR (planned_start_at IS NOT NULL AND expected_exit_at IS NOT NULL AND expected_exit_at > planned_start_at)` [name: 'chk_group_planned_interval']


    // Фактическое завершение не может предшествовать
    // фактическому началу похода.

    `actual_exit_at IS NULL OR (actual_start_at IS NOT NULL AND actual_exit_at >= actual_start_at)` [name: 'chk_group_actual_interval']


    // forming, scheduled, cancelled:
    // поход еще не начинался.
    //
    // on_route:
    // поход начался, но не завершился.
    //
    // completed, evacuated:
    // поход начался и завершился.

    `((status IN ('forming', 'scheduled', 'cancelled') AND actual_start_at IS NULL AND actual_exit_at IS NULL) OR (status = 'on_route' AND actual_start_at IS NOT NULL AND actual_exit_at IS NULL) OR (status IN ('completed', 'evacuated') AND actual_start_at IS NOT NULL AND actual_exit_at IS NOT NULL))` [name: 'chk_group_status_dates']


    // Перед планированием и отправкой должны быть
    // назначены старший и плановые даты.

    `(status IN ('forming', 'cancelled') OR (leader_certificate_no IS NOT NULL AND planned_start_at IS NOT NULL AND expected_exit_at IS NOT NULL))` [name: 'chk_group_readiness']
  }

  Note: '''
    Одна запись — один конкретный групповой поход.

    Целевая вершина определяется через routes.peak_id.

    СТАТУСЫ:

    forming:
      Создается группа и набирается состав.
      Фактические даты отсутствуют.
      Старший и плановые даты могут отсутствовать.

    scheduled:
      Группа сформирована и поход запланирован.
      Старший и плановые даты обязательны.
      Фактические даты отсутствуют.

    on_route:
      Поход фактически начался.
      actual_start_at обязателен.
      actual_exit_at отсутствует.

    completed:
      Поход завершен без завершения всей группы
      эвакуацией.
      Обе фактические даты обязательны.
      Вершина могла быть покорена или не покорена.
      Частичная эвакуация участников допускается.

    evacuated:
      Весь поход завершился эвакуацией группы.
      Обе фактические даты обязательны.
      Должно существовать хотя бы одно
      соответствующее эвакуационное событие.

    cancelled:
      Поход отменен до фактического начала.
      Фактические даты отсутствуют.
      Старший и плановые даты могут отсутствовать.

    ПРАВИЛА, КОТОРЫЕ ТРЕБУЮТ ТРИГГЕРА ИЛИ БИЗНЕС-ЛОГИКИ:

    1. Перед переходом в scheduled и on_route
       должен существовать непустой состав группы.

    2. Старший должен входить в эту группу.
       Саму принадлежность обеспечивает составной FK.

    3. Перед началом похода участники должны
       удовлетворять правилам допуска клуба.

    4. После фактического начала похода запрещается
       изменять route_id.

    5. Запрещается обнулять или изменять
       actual_start_at с целью обойти защиту истории.

    6. После начала похода запрещается возвращать
       группу в forming, scheduled или cancelled.

    7. Для статуса evacuated должна существовать
       связанная эвакуация, завершившая поход.

    8. Частичная эвакуация не изменяет on_route,
       если остальные участники продолжают маршрут.

    9. Если весь поход прекращен эвакуацией,
       устанавливаются evacuated и actual_exit_at.

    10. Если поход завершился обычным выходом
        с маршрута, устанавливается completed.
        Успешное восхождение для этого не требуется.

    CHECK проверяет текущее состояние строки,
    но не допустимость перехода из старого состояния.

    Защита уже начатого или завершенного похода
    обеспечивается проверкой OLD/NEW в триггере
    либо бизнес-логикой с соответствующими
    ограничениями доступа.
  '''
}


// ============================================================
// 7. СОСТАВ ГРУПП
// ============================================================

Table group_members {
  group_id int [not null]

  climber_certificate_no varchar(50) [not null]

  added_at timestamp [not null]

  indexes {
    (group_id, climber_certificate_no) [pk]
    climber_certificate_no
  }

  Note: '''
    Исторический исходный состав группы.

    Связь групп и альпинистов M:N.

    Альпинист может участвовать в нескольких
    различных походах.

    В одной группе он учитывается только один раз.

    При добавлении нового участника необходимо
    проверить его активный статус.

    После фактического начала похода запрещается
    добавлять, удалять или заменять участников.

    Для защиты истории требуется триггер
    или бизнес-логика.

    Эвакуация участника не удаляет его из group_members.

    Текущий состав на маршруте вычисляется
    из исходного состава с исключением участников,
    уже зарегистрированных в evacuation_rescued
    для этой группы.

    Это вычисление использует допущение,
    что единственный учитываемый вид досрочного
    выбытия участника из продолжающегося похода
    — зарегистрированная эвакуация.

    Другие причины досрочного выбытия отдельно
    не моделируются в рамках данного задания.
  '''
}


// ============================================================
// 8. УСПЕШНЫЕ ВОСХОЖДЕНИЯ ГРУПП
// ============================================================

Table summit_ascents {
  id int [pk, increment]

  group_id int [not null, unique]

  reached_at timestamp [not null]

  notes text

  indexes {
    (id, group_id) [unique]
  }

  Note: '''
    Факт достижения целевой вершины группой.

    Одна группа — не более одного успешного
    восхождения в рамках данного похода.

    Уникальность group_id обеспечивает
    ограничение не более одной записи.

    Группа может существовать без summit_ascents,
    включая завершенный поход.

    Следовательно, завершенный поход не означает
    автоматически успешного покорения вершины.

    Для каждого успешного восхождения должен
    существовать хотя бы один участник
    в summit_participants.

    Это правило требует триггера или бизнес-логики:
    внешний ключ в обратную сторону отсутствует.

    reached_at должен быть:

    - не раньше actual_start_at соответствующей группы;
    - не позже actual_exit_at, если она известна.

    Группа должна иметь фактически начатый поход.

    Эти условия требуют межтабличной проверки.

    Целевая вершина определяется через маршрут группы.

    route_id группы и peak_id маршрута после начала
    похода защищаются от изменения внешними правилами.

    Составной уникальный индекс (id, group_id)
    сохранен для внешнего ключа summit_participants.
  '''
}


// ============================================================
// 9. УЧАСТНИКИ, ЛИЧНО ДОСТИГШИЕ ВЕРШИНЫ
// ============================================================

Table summit_participants {
  ascent_id int [not null]

  group_id int [not null]

  climber_certificate_no varchar(50) [not null]

  indexes {
    (ascent_id, climber_certificate_no) [pk]
    (group_id, climber_certificate_no)
  }

  Note: '''
    Фиксирует личное достижение вершины альпинистом.

    Не каждый участник группы обязательно
    становится участником успешного восхождения.

    Составные внешние ключи обеспечивают:

    1. Восхождение относится к указанной группе.

    2. Альпинист входит в состав этой группы.

    Один альпинист учитывается не более одного раза
    в рамках одного восхождения.

    ВРЕМЕННОЕ ПРАВИЛО ПРИ ЭВАКУАЦИИ:

    Если участник был эвакуирован из этого похода,
    он может числиться покорителем вершины только
    при условии:

      summit_ascents.reached_at <
      времени его эвакуации.

    Если эвакуация произошла раньше достижения
    вершины группой, участник не может быть внесен
    в summit_participants.

    Личное восхождение, состоявшееся до эвакуации,
    сохраняется независимо от дальнейшего статуса
    альпиниста или группы.

    Это межтабличное ограничение выполняется
    триггером или бизнес-логикой.

    История восхождений доступна через:

    summit_participants
      -> summit_ascents
      -> climbing_groups
      -> routes
      -> peaks.
  '''
}


// ============================================================
// 10. ЭВАКУАЦИОННЫЕ СОБЫТИЯ
// ============================================================

Table evacuations {
  id int [pk, increment]

  group_id int [not null]

  occurred_at timestamp [not null]

  reason text

  indexes {
    group_id
    occurred_at
    (id, group_id) [unique]
  }

  Note: '''
    Эвакуационное событие на маршруте.

    Одна группа может иметь несколько
    эвакуационных событий.

    occurred_at хранит дату и время события.

    Список спасенных хранится отдельно
    в evacuation_rescued.

    ВРЕМЕННЫЕ ПРАВИЛА:

    1. Эвакуация допустима только в походе,
       который фактически начался.

    2. occurred_at >= actual_start_at группы.

    3. Если actual_exit_at определен:
       occurred_at <= actual_exit_at.

    4. Нельзя добавлять новое событие вне
       фактического интервала похода.

    Это проверки между таблицами.
    Обычный CHECK здесь недостаточен.

    ЧАСТИЧНАЯ ЭВАКУАЦИЯ:

    Если эвакуированы только отдельные участники,
    а остальные продолжают поход, статус группы
    остается on_route.

    Эвакуированные участники остаются в историческом
    исходном составе group_members.

    Если эвакуацией завершен весь поход,
    группе присваивается evacuated,
    устанавливается actual_exit_at.

    Одна лишь запись в evacuations не означает,
    что весь поход завершился.

    Для зарегистрированной эвакуации должен
    существовать хотя бы один спасенный участник
    в evacuation_rescued.

    Наличие спасенных проверяется триггером
    или бизнес-логикой при фиксации события.

    Для рейтинга опасности вершины одна группа
    с несколькими эвакуациями учитывается
    как один поход с эвакуацией.

    Уникальность (id, group_id) необходима
    для составного внешнего ключа списка спасенных.
  '''
}


// ============================================================
// 11. СПАСЕННЫЕ ПРИ ЭВАКУАЦИИ
// ============================================================

Table evacuation_rescued {
  evacuation_id int [not null]

  group_id int [not null]

  climber_certificate_no varchar(50) [not null]

  indexes {
    (evacuation_id, climber_certificate_no) [pk]

    (group_id, climber_certificate_no) [unique]
  }

  Note: '''
    Список спасенных альпинистов.

    Составные внешние ключи обеспечивают:

    1. Эвакуация относится к указанной группе.

    2. Спасенный альпинист входит в исходный
       состав указанной группы.

    Первичный ключ запрещает повторное указание
    альпиниста в одном эвакуационном событии.

    Уникальный индекс (group_id, climber_certificate_no)
    запрещает повторную регистрацию эвакуации
    одного и того же альпиниста в рамках похода.

    Это соответствует допущению:
    эвакуированный участник в этот поход
    на маршрут не возвращается.

    Поскольку пара (group_id, climber_certificate_no)
    уникальна и здесь, и в group_members,
    связь по этой паре объявлена как 1:1.

    При этом спасение необязательно для
    каждого участника группы.

    Наличие записи в этой таблице означает
    зарегистрированное спасение при эвакуации.

    Отсутствие записи не означает гибель альпиниста.

    Гибель учитывается отдельно через
    climbers.status = deceased.

    Запрет личного восхождения после эвакуации
    обеспечивается межтабличной проверкой
    с summit_participants и summit_ascents.

    Исторические записи эвакуации не удаляются.
  '''
}


// ============================================================
// ВНЕШНИЕ КЛЮЧИ И СВЯЗИ
// ============================================================
//
// Для всех связей используется запрет каскадного удаления.
// restrict защищает существующие ссылки, но сам по себе
// не предотвращает последовательное удаление всех
// зависимых исторических строк.
// Полный запрет такого удаления обеспечивается отдельно.


// ------------------------------------------------------------
// Вершина — маршруты: 1:N
// ------------------------------------------------------------

Ref: routes.peak_id > peaks.id [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Маршруты — населенные пункты: M:N
// ------------------------------------------------------------

Ref: route_endpoints.route_id > routes.id [delete: restrict, update: restrict]

Ref: route_endpoints.settlement_id > settlements.id [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Маршрут — групповые походы: 1:N
// ------------------------------------------------------------

Ref: climbing_groups.route_id > routes.id [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Группы — альпинисты: M:N
// ------------------------------------------------------------

Ref: group_members.group_id > climbing_groups.id [delete: restrict, update: restrict]

Ref: group_members.climber_certificate_no > climbers.certificate_no [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Старший группы
// ------------------------------------------------------------
//
// Составной FK гарантирует принадлежность старшего
// именно той группе, которой он руководит.
//
// Пока старший не назначен, leader_certificate_no = NULL.
// После назначения оба поля составного FK определены.
//
// Обязательность назначения перед scheduled/on_route
// обеспечивается CHECK climbing_groups.
//
// Корректность последовательности формирования:
// сначала группа, затем участники, затем старший.

Ref: climbing_groups.(id, leader_certificate_no) > group_members.(group_id, climber_certificate_no) [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Группа — успешное восхождение: 1:0..1
// ------------------------------------------------------------
//
// ИСПРАВЛЕНИЕ ПРЕДУПРЕЖДЕНИЙ DBML:
//
// Используется ровно одно объявление связи.
// Оператор "-" согласован с unique у group_id.
//
// В отдельном объявлении 1:1 внешний ключ справа:
// summit_ascents.group_id.
//
// Каждая запись summit_ascents обязательно имеет группу.
// Группа может не иметь успешного восхождения.

Ref: climbing_groups.id - summit_ascents.group_id [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Восхождения — покорители: 1:N
// ------------------------------------------------------------
//
// Составной FK сохраняет group_id и проверяет,
// что ascent_id принадлежит указанной группе.
//
// Родительский ключ:
// summit_ascents.(id, group_id) UNIQUE.

Ref: summit_participants.(ascent_id, group_id) > summit_ascents.(id, group_id) [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Состав группы — личные покорения: 1:N
// ------------------------------------------------------------
//
// Альпинист может покорить вершину только
// в составе соответствующей группы.
//
// Родительский ключ:
// group_members.(group_id, climber_certificate_no) PK.

Ref: summit_participants.(group_id, climber_certificate_no) > group_members.(group_id, climber_certificate_no) [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Группа — эвакуационные события: 1:N
// ------------------------------------------------------------

Ref: evacuations.group_id > climbing_groups.id [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Эвакуация — спасенные: 1:N
// ------------------------------------------------------------
//
// Составной FK исключает указание чужой группы.
//
// Родительский ключ:
// evacuations.(id, group_id) UNIQUE.

Ref: evacuation_rescued.(evacuation_id, group_id) > evacuations.(id, group_id) [delete: restrict, update: restrict]


// ------------------------------------------------------------
// Участник группы — факт его спасения: 1:0..1
// ------------------------------------------------------------
//
// Уникальность (group_id, climber_certificate_no)
// в evacuation_rescued позволяет использовать "-".
// Внешний ключ находится справа.
//
// Один участник может не иметь записи о спасении.
// В рамках одного похода допускается не более одного
// зарегистрированного спасения этого участника.
//
// Составной FK гарантирует принадлежность спасенного
// именно соответствующей группе.

Ref: group_members.(group_id, climber_certificate_no) - evacuation_rescued.(group_id, climber_certificate_no) [delete: restrict, update: restrict]


// ============================================================
// ДОПОЛНИТЕЛЬНЫЕ ПРАВИЛА ЦЕЛОСТНОСТИ
// ============================================================
//
// Следующие правила НЕ обеспечиваются приведенными
// обычными CHECK и FK. Для них необходимы триггеры
// или бизнес-логика с соответствующими правами доступа.
//
// ПРАВИЛО A. ФОРМИРОВАНИЕ ГРУППЫ
//
// - Новый участник должен быть активным.
// - Перед отправкой состав группы непустой.
// - Назначенный старший является участником группы.
// - Маршрут имеет стартовые и конечные пункты.
// - Альпинист не участвует одновременно в двух
//   пересекающихся по времени походах.
// - После начала похода состав и старший фиксируются.
//
// ПРАВИЛО B. ВРЕМЕННЫЕ ГРАНИЦЫ СОБЫТИЙ
//
// Для summit_ascents и evacuations:
//
// - связанный поход фактически начался;
// - событие не раньше actual_start_at;
// - событие не позже actual_exit_at, если дата известна.
//
// Эти условия проверяются не только при добавлении события,
// но и при изменении фактических дат связанного похода.
//
// ПРАВИЛО C. УСПЕШНОЕ ВОСХОЖДЕНИЕ
//
// - У восхождения минимум один summit_participants.
// - Каждый покоритель принадлежит исходной группе.
// - Участник не может быть покорителем после эвакуации.
// - Изменение времени восхождения не должно нарушать
//   временные границы и историю эвакуаций.
//
// Для создания восхождения и его участников допускается
// одна транзакция с проверкой при ее завершении.
//
// ПРАВИЛО D. ЭВАКУАЦИЯ
//
// - У зарегистрированной эвакуации есть спасенные.
// - Спасенные принадлежат этой группе.
// - Спасенный не эвакуируется повторно в том же походе.
// - Уже спасенный не возвращается на маршрут.
// - Частичная эвакуация не завершает весь поход.
// - Полное завершение похода эвакуацией требует
//   статуса evacuated и фактической даты выхода.
// - Изменение времени эвакуации не должно делать
//   ранее записанное личное восхождение невозможным.
//
// ПРАВИЛО E. НЕИЗМЕНЯЕМОСТЬ ИСТОРИИ
//
// После первого фактического начала похода:
//
// - climbing_groups.route_id нельзя изменить;
// - исходный состав group_members нельзя изменить;
// - actual_start_at нельзя обнулить для обхода правил;
// - маршрут нельзя переназначить другой вершине;
// - использованные характеристики маршрута фиксируются;
// - набор route_endpoints исторического маршрута фиксируется;
// - исторически значимые характеристики peaks защищаются;
// - записи личных восхождений и спасений сохраняются.
//
// Данные правила должны учитывать предыдущее состояние
// записи и наличие связанных начатых походов.
//
// Обычные CHECK и FK этого не обеспечивают.
//
// Для нового варианта маршрута создается новая запись,
// а не редактируется исторически использованный маршрут.


// ============================================================
// ПРОВЕРКА ОПЕРАЦИЙ ИЗ ИСХОДНОГО ЗАДАНИЯ
// ============================================================
//
// ОПЕРАЦИЯ 1. Внесение новой вершины
//
// Таблица: peaks.
//
// Обязательны страна, название и положительная высота.
//
//
// ОПЕРАЦИЯ 2. Создание нового маршрута
//
// Таблицы: routes, route_endpoints, settlements.
//
// Обязательны вершина, название, положительная
// продолжительность, категория сложности.
//
// Маршруту назначаются минимум один стартовый
// и минимум один конечный населенный пункт.
//
//
// ОПЕРАЦИЯ 3. Постановка на учет альпиниста
//
// Таблица: climbers.
//
// Сертификат уникален.
// Новый альпинист имеет статус active.
//
//
// ОПЕРАЦИЯ 4. Перевод в неактивное состояние
//
// Таблица: climbers.
//
// status = departed или deceased.
// status_changed_at фиксирует дату изменения.
//
// История участия и покорения вершин сохраняется.
//
//
// ОПЕРАЦИЯ 5. Формирование группы, назначение старшего
//
// Таблицы: climbing_groups, group_members.
//
// Последовательность:
// - создать группу со статусом forming;
// - добавить участников;
// - назначить старшего;
// - определить плановые даты;
// - перевести в scheduled;
// - при фактическом выходе перевести в on_route.
//
// Составной внешний ключ гарантирует,
// что старший принадлежит соответствующей группе.


// ============================================================
// ПРОВЕРКА ЧЕТЫРЕХ ОТЧЕТОВ
// ============================================================
//
// ОТЧЕТ 1. АКТИВНЫЕ ГРУППЫ НА МАРШРУТЕ
//
// Основные таблицы:
//
// climbing_groups
// routes
// group_members
// climbers
// evacuation_rescued
//
// Условие активности:
// climbing_groups.status = 'on_route'.
//
// Выводятся:
// - название группы;
// - старший;
// - маршрут;
// - фактически оставшиеся на маршруте участники;
// - ожидаемое время выхода expected_exit_at.
//
// Принято следующее определение состава в оперативном
// отчете:
//
// ТЕКУЩИЙ СОСТАВ =
// исходный group_members
// минус участники, уже указанные в evacuation_rescued
// для данного group_id.
//
// Таким образом, после частичной эвакуации
// отчет не показывает спасенных как продолжающих маршрут.
//
// При необходимости исходный состав группы
// также доступен из group_members без фильтрации.
//
// Данный расчет фактического состава использует допущение,
// что другие виды досрочного выбытия не моделируются.
//
// Группировка осуществляется по climbing_groups.id.
// Старший определяется через leader_certificate_no.
//
//
// ОТЧЕТ 2. ИСТОРИЯ ПОКОРЕНИЯ ВЕРШИН АЛЬПИНИСТОМ
//
// Таблицы:
//
// climbers
// summit_participants
// summit_ascents
// climbing_groups
// routes
// peaks
//
// Для выбранного certificate_no отображаются:
// - вершина;
// - страна;
// - высота;
// - дата и время достижения вершины;
// - группа;
// - маршрут;
// - категория сложности.
//
// Учитываются только личные успешные восхождения.
//
// Альпинист, участвовавший в походе, но не достигший
// вершины, не получает запись о покорении.
//
// Восхождение до эвакуации сохраняется.
//
// История доступна для активных, ушедших
// и погибших альпинистов.
//
//
// ОТЧЕТ 3. САМЫЕ ОПАСНЫЕ ГОРЫ — ТОП-5
//
// Таблицы:
//
// peaks
// routes
// climbing_groups
// evacuations
//
// Принятый критерий опасности:
//
// started_groups — количество уникальных групповых
// походов к вершине с actual_start_at IS NOT NULL.
//
// evacuated_groups — количество таких уникальных
// групповых походов, в которых зарегистрирована
// хотя бы одна эвакуация.
//
// danger_rate = evacuated_groups / started_groups.
//
// Частичная эвакуация тоже считается происшествием.
//
// Несколько эвакуационных событий одной группы
// учитываются как один поход с эвакуацией.
//
// В рейтинг включаются вершины, у которых
// started_groups > 0.
//
// Сортировка:
//
// 1. danger_rate по убыванию;
// 2. evacuated_groups по убыванию;
// 3. peaks.id по возрастанию для устойчивого порядка.
//
// Выводятся первые пять вершин.
//
// Успешное достижение вершины не отменяет
// ранее зарегистрированную эвакуацию.
//
//
// ОТЧЕТ 4. САМЫЕ КРУТЫЕ АЛЬПИНИСТЫ
//
// Таблицы:
//
// climbers
// summit_participants
// summit_ascents
// climbing_groups
// routes
// peaks
//
// Принятый основной показатель:
//
// total_ascents — количество личных
// успешных восхождений альпиниста.
//
// Дополнительные показатели:
//
// unique_peaks — число разных покоренных вершин;
//
// max_elevation — максимальная высота
// покоренной вершины;
//
// total_elevation — сумма высот вершин
// по всем успешным личным восхождениям;
//
// difficulty_category — сложность маршрутов,
// по которым совершались успешные восхождения.
//
// Основная сортировка рейтинга:
//
// 1. total_ascents по убыванию;
// 2. unique_peaks по убыванию;
// 3. max_elevation по убыванию;
// 4. certificate_no по возрастанию.
//
// Сложность маршрутов доступна для дополнительной
// статистики и альтернативных рейтингов.
//
// Можно выводить полный список или топ-N.
//
// Альпинисты без успешных восхождений
// включаются в полный список с нулевыми показателями.
//
// Неактивные альпинисты из истории не исключаются.


// ============================================================
// ИТОГОВЫЕ ГАРАНТИИ СХЕМЫ
// ============================================================
//
// DBML непосредственно задает:
//
// - первичные ключи;
// - внешние ключи;
// - составные внешние ключи;
// - уникальные ограничения;
// - обязательность полей;
// - допустимые значения Enum;
// - CHECK для простых числовых и временных правил;
// - CHECK согласованности статуса с датами;
// - корректную связь группы и восхождения 1:0..1;
// - запрет повторного спасения участника в одном походе.
//
// Триггеры или бизнес-логика дополнительно необходимы для:
//
// - проверки существования участников и пунктов маршрута;
// - проверки допустимых переходов состояний;
// - контроля временных границ связанных событий;
// - проверки наличия участников успешного восхождения;
// - защиты личного восхождения после эвакуации;
// - запрета изменения исторических маршрутов и вершин;
// - защиты исходного состава группы;
